-- MERGE FILA
CREATE OR ALTER PROCEDURE dbo.usp_merge_fila
AS
BEGIN
    SET NOCOUNT ON;

    MERGE dbo.fila AS t
    USING (
        SELECT
            TRY_CAST(id_fila AS INT)              AS id_fila,
            cd_fila,
            nm_fila,
            ds_descricao,
            CASE WHEN UPPER(fl_ativo) IN ('1','TRUE','S','SIM','Y','YES') THEN 1 ELSE 0 END AS fl_ativo,
            TRY_CAST(dt_inclusao AS DATETIME2(0)) AS dt_inclusao,
            TRY_CAST(dt_atualizacao AS DATETIME2(0)) AS dt_atualizacao,
            nm_sistema_origem,
            cd_registro_origem
        FROM dbo.stg_fila
        WHERE TRY_CAST(id_fila AS INT) IS NOT NULL
    ) AS s
    ON t.id_fila = s.id_fila

    WHEN MATCHED THEN UPDATE SET
        t.cd_fila            = s.cd_fila,
        t.nm_fila            = s.nm_fila,
        t.ds_descricao       = s.ds_descricao,
        t.fl_ativo           = s.fl_ativo,
        t.dt_atualizacao     = s.dt_atualizacao,
        t.nm_sistema_origem  = s.nm_sistema_origem,
        t.cd_registro_origem = s.cd_registro_origem

    WHEN NOT MATCHED BY TARGET THEN
        INSERT (id_fila, cd_fila, nm_fila, ds_descricao, fl_ativo, dt_inclusao, dt_atualizacao, nm_sistema_origem, cd_registro_origem)
        VALUES (s.id_fila, s.cd_fila, s.nm_fila, s.ds_descricao, s.fl_ativo, s.dt_inclusao, s.dt_atualizacao, s.nm_sistema_origem, s.cd_registro_origem);
END
GO
